static long xsys6(long nr, long a0, long a1, long a2,
		  long a3, long a4, long a5)
{
	long ret;
	register long r10 __asm__("r10") = a3;
	register long r8 __asm__("r8") = a4;
	register long r9 __asm__("r9") = a5;

	__asm__ volatile (
		"syscall"
		: "=a"(ret)
		: "a"(nr), "D"(a0), "S"(a1), "d"(a2),
		  "r"(r10), "r"(r8), "r"(r9)
		: "rcx", "r11", "memory");
	return ret;
}

static long xsys3(long nr, long a0, long a1, long a2)
{
	return xsys6(nr, a0, a1, a2, 0, 0, 0);
}

static void memzero(void *ptr, size_t len)
{
	unsigned char *p = ptr;

	while (len--)
		*p++ = 0;
}

static void xclose(int fd)
{
	if (fd >= 0)
		xsys3(NR_CLOSE, fd, 0, 0);
}

static void xshutdown(int fd)
{
	if (fd >= 0)
		xsys3(NR_SHUTDOWN, fd, SHUT_RDWR_, 0);
}

static long xbpf(long cmd, union bpf_attr *attr, long size)
{
	return xsys3(NR_BPF, cmd, (long)attr, size);
}

static long xsocket(long family, long type, long proto)
{
	return xsys3(NR_SOCKET, family, type, proto);
}

static int map_syz_region(void)
{
	long ret;

	ret = xsys6(NR_MMAP, SYZ_BASE_ADDR, SYZ_MAP_LEN,
		    PROT_READ_ | PROT_WRITE_,
		    MAP_PRIVATE_ | MAP_ANON_ | MAP_FIXED_ | MAP_NORESERVE_,
		    -1, 0);
	if (ret != (long)SYZ_BASE_ADDR)
		return -1;

	*(volatile unsigned char *)(uintptr_t)SYZ_SEND_ADDR = 0x01;
	return 0;
}

static int create_sockmap(void)
{
	union bpf_attr *attr = (union bpf_attr *)(uintptr_t)SYZ_MAP_CREATE;

	memzero(attr, sizeof(*attr));
	attr->map_type = BPF_MAP_TYPE_SOCKMAP_;
	attr->key_size = SOCKMAP_KEY_SIZE;
	attr->value_size = SOCKMAP_VAL_SIZE;
	attr->max_entries = SOCKMAP_ENTRIES;

	return (int)xbpf(BPF_CMD_MAP_CREATE, attr, MAP_CREATE_SIZE);
}

static int load_sk_skb_pass_prog(void)
{
	static const char license[] = "GPL";
	struct bpf_insn insns[] = {
		BPF_ALU64_IMM(BPF_MOV, BPF_REG_0, SK_PASS_),
		BPF_EXIT_INSN(),
	};
	union bpf_attr *attr = (union bpf_attr *)(uintptr_t)SYZ_PROG_LOAD_ARG;

	memzero(attr, sizeof(*attr));
	attr->prog_type = BPF_PROG_TYPE_SK_SKB_;
	attr->insn_cnt = ARRAY_SIZE(insns);
	attr->insns = (uint64_t)(uintptr_t)insns;
	attr->license = (uint64_t)(uintptr_t)license;

	return (int)xbpf(BPF_CMD_PROG_LOAD, attr, PROG_LOAD_SIZE);
}

static int attach_sockmap_prog(int map_fd, int prog_fd, int attach_type)
{
	uintptr_t addr = attach_type == BPF_ATTACH_STREAM_PARSER ?
		SYZ_ATTACH_PARSER : SYZ_ATTACH_VERDICT;
	union bpf_attr *attr = (union bpf_attr *)addr;

	memzero(attr, sizeof(*attr));
	attr->target_fd = map_fd;
	attr->attach_bpf_fd = prog_fd;
	attr->attach_type = attach_type;

	return (int)xbpf(BPF_CMD_PROG_ATTACH, attr, PROG_ATTACH_SIZE);
}

static int update_sockmap(int map_fd, int sock_fd)
{
	union bpf_attr *attr = (union bpf_attr *)(uintptr_t)SYZ_MAP_UPDATE;
	uint32_t *key = (uint32_t *)(uintptr_t)SYZ_MAP_KEY;
	int *value = (int *)(uintptr_t)SYZ_MAP_VALUE;

	memzero(attr, sizeof(*attr));
	*key = 0;
	*value = sock_fd;
	attr->map_fd = map_fd;
	attr->key = SYZ_MAP_KEY;
	attr->value = SYZ_MAP_VALUE;
	attr->flags = BPF_ANY_;

	return (int)xbpf(BPF_CMD_MAP_UPDATE_ELEM, attr, MAP_UPDATE_SIZE);
}

static int create_tcp_pair(int *rx_fd, int *tx_fd)
{
	struct sockaddr_in addr = {};
	uint32_t addr_len = sizeof(addr);
	long listener;
	long tx;
	long rx;

	listener = xsocket(AF_INET_, SOCK_STREAM_, IPPROTO_IP_);
	if (listener < 0)
		return -1;

	addr.sin_family = AF_INET_;
	addr.sin_port = 0;
	addr.sin_addr.s_addr = LOOPBACK_ADDR_BE;

	if (xsys3(NR_BIND, listener, (long)&addr, sizeof(addr)) < 0)
		goto fail;
	if (xsys3(NR_LISTEN, listener, 1, 0) < 0)
		goto fail;
	if (xsys3(NR_GETSOCKNAME, listener, (long)&addr, (long)&addr_len) < 0)
		goto fail;

	tx = xsocket(AF_INET_, SOCK_STREAM_, IPPROTO_IP_);
	if (tx < 0)
		goto fail;
	if (xsys3(NR_CONNECT, tx, (long)&addr, sizeof(addr)) < 0) {
		xclose((int)tx);
		goto fail;
	}

	rx = xsys3(NR_ACCEPT, listener, 0, 0);
	if (rx < 0) {
		xclose((int)tx);
		goto fail;
	}

	xclose((int)listener);
	*rx_fd = (int)rx;
	*tx_fd = (int)tx;
	return 0;

fail:
	xclose((int)listener);
	return -1;
}

static int syz_socket_pair(int family, int type, int proto, int *rx_fd, int *tx_fd)
{
	int *out = (int *)(uintptr_t)SYZ_SOCKPAIR_OUT;
	long rc;

	memzero(out, sizeof(int) * 2);
	out[0] = -1;
	out[1] = -1;

	rc = xsys6(NR_SOCKETPAIR, family, type, proto, SYZ_SOCKPAIR_OUT, 0, 0);
	if (rc == 0) {
		*rx_fd = out[0];
		*tx_fd = out[1];
		return 0;
	}

	/*
	 * syzkaller's syz_socket_pair pseudo-call creates a connected TCP pair.
	 * Linux socketpair(AF_INET, SOCK_STREAM, 0) returns -EOPNOTSUPP.
	 */
	return create_tcp_pair(rx_fd, tx_fd);
}

static void teardown_fds(int iter, int map_fd, int prog_fd, int rx_fd, int tx_fd)
{
	switch (iter & 3) {
	case 0:
		xshutdown(tx_fd);
		xshutdown(rx_fd);
		xclose(tx_fd);
		xclose(rx_fd);
		xclose(prog_fd);
		xclose(map_fd);
		break;
	case 1:
		xshutdown(rx_fd);
		xshutdown(tx_fd);
		xclose(rx_fd);
		xclose(tx_fd);
		xclose(map_fd);
		xclose(prog_fd);
		break;
	case 2:
		xshutdown(tx_fd);
		xclose(map_fd);
		xclose(prog_fd);
		xclose(tx_fd);
		xshutdown(rx_fd);
		xclose(rx_fd);
		break;
	default:
		xshutdown(rx_fd);
		xclose(rx_fd);
		xclose(map_fd);
		xshutdown(tx_fd);
		xclose(tx_fd);
		xclose(prog_fd);
		break;
	}
}

static int run_one_program(int iter)
{
	int map_fd = -1;
	int prog_fd = -1;
	int rx_fd = -1;
	int tx_fd = -1;
	int ret = 1;

	map_fd = create_sockmap();
	if (map_fd < 0)
		goto out;

	prog_fd = load_sk_skb_pass_prog();
	if (prog_fd < 0)
		goto out;

	if (attach_sockmap_prog(map_fd, prog_fd, BPF_ATTACH_STREAM_VERDICT) < 0)
		goto out;
	if (attach_sockmap_prog(map_fd, prog_fd, BPF_ATTACH_STREAM_PARSER) < 0)
		goto out;
	if (syz_socket_pair(AF_INET_, SOCK_STREAM_, IPPROTO_IP_, &rx_fd, &tx_fd) < 0)
		goto out;
	if (update_sockmap(map_fd, rx_fd) < 0)
		goto out;

	xsys6(NR_SENDTO, tx_fd, SYZ_SEND_ADDR, RAW_SEND_LEN, 0, 0, 0);
	ret = 0;

out:
	teardown_fds(iter, map_fd, prog_fd, rx_fd, tx_fd);
	return ret;
}

static int run_loop(int worker)
{
	int i;

	for (i = 0; i < PROGRAM_REPEAT; i++) {
		if (run_one_program(i * PROGRAM_PROCS + worker) != 0)
			return 1;
	}

	return 0;
}

int main(void)
{
	long pids[PROGRAM_PROCS - 1];
	int i;
	int status;

	if (map_syz_region() < 0)
		return 1;

	for (i = 0; i < PROGRAM_PROCS - 1; i++) {
		long pid = xsys6(NR_CLONE, SIGCHLD_, 0, 0, 0, 0, 0);

		if (pid < 0)
			return 1;
		if (pid == 0)
			xsys3(NR_EXIT, run_loop(i + 1), 0, 0);
		pids[i] = pid;
	}

	status = run_loop(0);

	for (i = 0; i < PROGRAM_PROCS - 1; i++)
		xsys6(NR_WAIT4, pids[i], 0, 0, 0, 0, 0);

	return status;
}
